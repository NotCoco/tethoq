import { readFile, readdir, unlink } from "node:fs/promises"
import { homedir } from "node:os"
import { join } from "node:path"
import { createConnection } from "node:net"
import { tool } from "@opencode-ai/plugin"

const configuredRuntimePath = process.env.UAR_MESH_RUNTIME
const legacyRuntimePath = join(homedir(), ".tethoq", "mesh-tool-runtime.json")
const runtimeDirectory = join(homedir(), ".tethoq", "mesh-runtimes")
const runtimeResponseTimeoutMs = 2_000
const runtimeActiveTimeoutMs = 30_000
const runtimeRecoveryTimeoutMs = 30_000
// The only EYES sentence a provider model may see when the visual-support pipe
// itself is unreachable. Real usage, auth, and model failures keep their own
// terminal errors; transport noise must never leak a raw pipe diagnostic.
const eyesTransportFailure = "EYES could not inspect the image. Try again or choose another EYES model."

class RuntimeConnectError extends Error {
  constructor(error, mayHaveRun = false, canForget = false) {
    super(mayHaveRun
      ? "Tethoq lost the tool connection before its result arrived. The action may have been accepted; check the task or child state before sending it again."
      : "The Tethoq tool runtime is reconnecting. Try again shortly.", { cause: error })
    this.name = "RuntimeConnectError"
    this.code = mayHaveRun ? "TOOL_DELIVERY_UNKNOWN" : "RUNTIME_UNAVAILABLE"
    this.mayHaveRun = mayHaveRun
    this.canForget = canForget
  }
}

class RuntimeTaskNotOwnedError extends Error {
  constructor(message) {
    super(message || "This Tethoq runtime does not own the task")
    this.name = "RuntimeTaskNotOwnedError"
    this.code = "TASK_NOT_OWNED_HERE"
  }
}

class RuntimeEyesNotConfiguredError extends Error {
  constructor(message) {
    super(message || "No visual-support model is configured for this session.")
    this.name = "RuntimeEyesNotConfiguredError"
    this.code = "EYES_NOT_CONFIGURED"
  }
}

async function call(sessionID, name, input, callId) {
  // OpenCode caches the tool module for the life of its server. Resolve the
  // installed transport per invocation so app updates also reach active servers.
  if (!import.meta.url.startsWith("file:")) return await invoke(sessionID, name, input, callId)
  const url = new URL(import.meta.url)
  url.search = ""
  url.hash = ""
  const source = await readFile(url, "utf8")
  // Import the dependency-free transport by content. Bun can retain the old
  // file module even when its URL query changes; a data URL identifies the bytes.
  const transport = source.replace(/^import \{ tool \} from [^\r\n]+\r?\n/mu, "")
    .replace(/^export const [\s\S]*/mu, "")
  const current = await import(`data:text/javascript;base64,${Buffer.from(transport).toString("base64")}`)
  return await current.invoke(sessionID, name, input, callId)
}

// A plain function export is not a provider tool; only tool definitions register.
export async function invoke(sessionID, name, input, callId) {
  let deadline
  const waitDeadline = name === "mesh_wait" ? Date.now() + Math.min(300, Math.max(1, input.timeout_seconds ?? 120)) * 1000 : undefined
  const attempted = new Set()
  let lastError
  let configurationError
  let taskNotOwnedError
  do {
    const runtimes = await matchingRuntimes(name)
    for (const runtime of runtimes) {
      const key = JSON.stringify([runtime.hostId, runtime.pipePath, runtime.token])
      if (attempted.has(key)) continue
      attempted.add(key)
      try {
        return await callRuntime(runtime, sessionID, name, waitDeadline === undefined ? input : { ...input, timeout_seconds: Math.max(1, Math.ceil((waitDeadline - Date.now()) / 1000)) }, callId)
      } catch (error) {
        if (error instanceof RuntimeEyesNotConfiguredError) { configurationError = error; continue }
        if (error instanceof RuntimeTaskNotOwnedError) {
          if (!taskNotOwnedError) taskNotOwnedError = error
          continue
        }
        if (!(error instanceof RuntimeConnectError) || error.mayHaveRun) throw error
        lastError = error
        deadline ??= Math.min(Date.now() + runtimeRecoveryTimeoutMs, waitDeadline === undefined ? Infinity : waitDeadline + runtimeRecoveryTimeoutMs)
        if (error.canForget) await forgetDeadRuntime(runtime)
      }
    }
    deadline ??= Date.now() + runtimeRecoveryTimeoutMs
    if (configurationError || taskNotOwnedError || Date.now() >= deadline) break
    await new Promise(resolve => setTimeout(resolve, 250))
  } while (Date.now() < deadline)
  throw configurationError || taskNotOwnedError || lastError || new Error("The Tethoq tool runtime is reconnecting. Try again shortly.")
}

async function forgetDeadRuntime(runtime) {
  if (Number.isInteger(runtime.pid) && runtime.pid > 0) {
    try { process.kill(runtime.pid, 0); return }
    catch (error) { if (error?.code !== "ESRCH") return }
  }
  // Crashed bridges leak descriptors that would otherwise be retried on every
  // later tool call. Remove the file that still names this exact unreachable
  // runtime; identity is rechecked so a recycled path is never deleted.
  try {
    const files = await readdir(runtimeDirectory)
    for (const file of files) {
      if (!file.endsWith(".json")) continue
      const path = join(runtimeDirectory, file)
      try {
        const candidate = JSON.parse(await readFile(path, "utf8"))
        if (candidate.hostId === runtime.hostId && candidate.pipePath === runtime.pipePath && candidate.token === runtime.token) {
          await unlink(path)
        }
      } catch {}
    }
  } catch {}
}

async function matchingRuntimes(name) {
  let paths = [legacyRuntimePath]
  try {
    paths.push(...(await readdir(runtimeDirectory)).filter((file) => file.endsWith(".json")).map((file) => join(runtimeDirectory, file)))
  } catch {}
  const discoveredRuntimes = []
  for (const path of paths) {
    try {
      const runtime = JSON.parse(await readFile(path, "utf8"))
      if (!Array.isArray(runtime.tools) || runtime.tools.includes(name)) discoveredRuntimes.push(runtime)
    } catch {}
  }
  discoveredRuntimes.sort((left, right) => String(right.startedAt || "").localeCompare(String(left.startedAt || "")))

  const runtimes = []
  if (configuredRuntimePath) {
    try {
      const runtime = JSON.parse(await readFile(configuredRuntimePath, "utf8"))
      if (!Array.isArray(runtime.tools) || runtime.tools.includes(name)) runtimes.push(runtime)
    } catch (error) {
      if (!isMissingFile(error)) throw error
    }
  }
  runtimes.push(...discoveredRuntimes)
  return dedupeRuntimes(runtimes)
}

function isMissingFile(error) {
  return error && typeof error === "object" && error.code === "ENOENT"
}

function dedupeRuntimes(runtimes) {
  const seen = new Set()
  return runtimes.filter((runtime) => {
    const key = [runtime.hostId, runtime.pipePath, runtime.token].map((value) => String(value || "")).join("\u0000")
    if (seen.has(key)) return false
    seen.add(key)
    return true
  })
}

function canRepeatRuntimeRead(name, input) {
  return ["mesh_list_children", "mesh_list_sessions", "mesh_wait", "mesh_read_result", "browser_get_state", "browser_inspect", "browser_inspect_all"].includes(name)
    || name === "tethoq_goal" && input.status === undefined
}

async function callRuntime(runtime, sessionID, name, input, callId) {
  const parentSessionId = [runtime.hostId, "opencode", sessionID].map(encodeURIComponent).join("/")
  return await new Promise((resolve, reject) => {
    const repeatable = canRepeatRuntimeRead(name, input)
    let socket
    let settled = false
    let sent = false
    let buffer = ""
    let deadline
    const finish = (error, value) => {
      if (settled) return
      settled = true
      clearTimeout(deadline)
      socket?.setTimeout(0)
      socket?.destroy()
      if (error) reject(error)
      else resolve(value)
    }
    const failed = (error) => finish(new RuntimeConnectError(error, sent && !repeatable,
      !sent && ["ENOENT", "ECONNREFUSED"].includes(error?.code)))
    try { socket = createConnection(runtime.pipePath) }
    catch (error) { failed(error); return }
    const waitSeconds = name === "mesh_wait" && typeof input.timeout_seconds === "number" ? Math.min(300, Math.max(1, input.timeout_seconds)) : 300
    deadline = setTimeout(() => failed(new Error("Tethoq tool execution timed out")), (waitSeconds + 30) * 1000)
    socket.setEncoding("utf8")
    socket.setTimeout(repeatable ? runtimeResponseTimeoutMs : runtimeActiveTimeoutMs, () => failed(new Error("Tethoq runtime response timed out")))
    socket.once("error", failed)
    socket.once("end", () => { if (!settled) failed(new Error("Tethoq runtime closed before replying")) })
    socket.once("close", () => { if (!settled) failed(new Error("Tethoq runtime disconnected")) })
    socket.on("data", (chunk) => {
      if (settled) return
      socket.setTimeout(runtimeActiveTimeoutMs)
      buffer += chunk
      if (buffer.length > 16 * 1024 * 1024) { failed(new Error("Tethoq tool response is too large")); return }
      const newline = buffer.indexOf("\n")
      if (newline < 0) return
      let response
      try { response = JSON.parse(buffer.slice(0, newline)) }
      catch (error) { failed(error); return }
      if (!response || typeof response.ok !== "boolean") { failed(new Error("Invalid Tethoq tool response")); return }
      if (!response.ok) {
        if (response.code === "TASK_NOT_OWNED_HERE") { finish(new RuntimeTaskNotOwnedError(response.error)); return }
        if (response.code === "EYES_NOT_CONFIGURED") { finish(new RuntimeEyesNotConfiguredError(response.error)); return }
        const error = new Error(response.error || "Tethoq tool failed")
        if (typeof response.code === "string") error.code = response.code
        finish(error)
        return
      }
      finish(undefined, JSON.stringify(response.result))
    })
    socket.once("connect", () => {
      sent = true
      socket.write(JSON.stringify({ token: runtime.token, parentSessionId, tool: name, input,
        context: { ...(callId ? { callId } : {}), lifecycleOwner: "provider" },
      }) + "\n")
    })
  })
}

export const tethoq_turn_support = tool({
  description: "Use only when private turn-scoped Tethoq guidance explicitly instructs you to call this tool. Do not infer a purpose or call it without that guidance.",
  args: { request: tool.schema.string().min(1).max(8000) },
  async execute(args, context) {
    try {
      return await call(context.sessionID, "tethoq_turn_support", args, context.callID || context.callId)
    } catch (error) {
      if (error?.code === "EYES_NOT_CONFIGURED") throw error
      if (["RUNTIME_UNAVAILABLE", "TOOL_DELIVERY_UNKNOWN", "TASK_NOT_OWNED_HERE"].includes(error?.code)) throw new Error(eyesTransportFailure)
      if (error instanceof Error && /No active Tethoq tool runtime supports this command/.test(error.message)) throw new Error(eyesTransportFailure)
      throw error
    }
  },
})
