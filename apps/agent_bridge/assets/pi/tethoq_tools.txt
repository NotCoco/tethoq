import { Type } from "typebox"
import { createConnection } from "node:net"
import { readFile, readdir, unlink } from "node:fs/promises"
import { homedir } from "node:os"
import { join } from "node:path"

const configuredRuntimePath = process.env.UAR_MESH_RUNTIME
const legacyRuntimePath = join(homedir(), ".tethoq", "mesh-tool-runtime.json")
const runtimeDirectory = join(homedir(), ".tethoq", "mesh-runtimes")
const runtimeResponseTimeoutMs = 2_000
// The only EYES sentence a provider model may see when the visual-support pipe
// itself is unreachable. Real usage, auth, and model failures keep their own
// terminal errors; transport noise must never leak a raw pipe diagnostic.
const eyesTransportFailure = "EYES could not inspect the image. Try again or choose another EYES model."

class RuntimeConnectError extends Error {
  constructor(error) {
    super(error instanceof Error ? error.message : String(error), { cause: error })
    this.name = "RuntimeConnectError"
  }
}

class RuntimeTaskNotOwnedError extends Error {
  constructor(message) {
    super(message || "This Tethoq runtime does not own the task")
    this.name = "RuntimeTaskNotOwnedError"
  }
}

async function call(sessionID, name, input, callId) {
  const runtimes = await matchingRuntimes(name)
  let lastError
  let taskNotOwnedError
  for (const runtime of runtimes) {
    try {
      return await callRuntime(runtime, sessionID, name, input, callId)
    } catch (error) {
      if (error instanceof RuntimeTaskNotOwnedError) {
        // A later dead descriptor's connect failure must not mask the
        // meaningful ownership miss once every runtime has been tried.
        if (!taskNotOwnedError) taskNotOwnedError = error
        continue
      }
      if (!(error instanceof RuntimeConnectError)) throw error
      lastError = error
      await forgetDeadRuntime(runtime)
    }
  }
  throw taskNotOwnedError || lastError || new Error("No active Tethoq tool runtime supports this command")
}

async function forgetDeadRuntime(runtime) {
  // Crashed bridges leak descriptors that would otherwise be retried on every
  // later tool call. Remove the file that still names this exact unreachable
  // runtime; identity is rechecked so a recycled path is never deleted.
  try {
    const files = await readdir(runtimeDirectory)
    for (const file of files) {
      if (!file.endsWith(".json")) continue
      const path = join(runtimeDirectory, file)
      try {
        const candidate = JSON.parse(await readFile(path, "utf8"))
        if (candidate.hostId === runtime.hostId && candidate.pipePath === runtime.pipePath && candidate.token === runtime.token) {
          await unlink(path)
        }
      } catch {}
    }
  } catch {}
}

async function matchingRuntimes(name) {
  let paths = [legacyRuntimePath]
  try {
    paths.push(...(await readdir(runtimeDirectory)).filter((file) => file.endsWith(".json")).map((file) => join(runtimeDirectory, file)))
  } catch {}
  const discoveredRuntimes = []
  for (const path of paths) {
    try {
      const runtime = JSON.parse(await readFile(path, "utf8"))
      if (!Array.isArray(runtime.tools) || runtime.tools.includes(name)) discoveredRuntimes.push(runtime)
    } catch {}
  }
  discoveredRuntimes.sort((left, right) => String(right.startedAt || "").localeCompare(String(left.startedAt || "")))

  const runtimes = []
  if (configuredRuntimePath) {
    try {
      const runtime = JSON.parse(await readFile(configuredRuntimePath, "utf8"))
      if (!Array.isArray(runtime.tools) || runtime.tools.includes(name)) runtimes.push(runtime)
    } catch (error) {
      if (!isMissingFile(error)) throw error
    }
  }
  runtimes.push(...discoveredRuntimes)
  return dedupeRuntimes(runtimes)
}

function isMissingFile(error) {
  return error && typeof error === "object" && error.code === "ENOENT"
}

function dedupeRuntimes(runtimes) {
  const seen = new Set()
  return runtimes.filter((runtime) => {
    const key = [runtime.hostId, runtime.pipePath, runtime.token].map((value) => String(value || "")).join("\u0000")
    if (seen.has(key)) return false
    seen.add(key)
    return true
  })
}

async function callRuntime(runtime, sessionID, name, input, callId) {
  const parentSessionId = [runtime.hostId, "pi", sessionID].map(encodeURIComponent).join("/")
  return await new Promise((resolve, reject) => {
    let socket
    try {
      socket = createConnection(runtime.pipePath)
    } catch (error) {
      reject(new RuntimeConnectError(error))
      return
    }
    let buffer = ""
    socket.setEncoding("utf8")
    socket.setTimeout(runtimeResponseTimeoutMs, () => {
      socket.destroy()
      reject(new RuntimeConnectError(new Error("Tethoq runtime response timed out")))
    })
    socket.once("error", (error) => reject(new RuntimeConnectError(error)))
    socket.on("data", (chunk) => {
      buffer += chunk
      const newline = buffer.indexOf("\n")
      if (newline < 0) return
      socket.setTimeout(0)
      socket.destroy()
      try {
        const response = JSON.parse(buffer.slice(0, newline))
        if (!response.ok) {
          if (response.code === "TASK_NOT_OWNED_HERE") throw new RuntimeTaskNotOwnedError(response.error)
          throw new Error(response.error || "Tethoq tool failed")
        }
        resolve(response.result)
      } catch (error) {
        reject(error)
      }
    })
    socket.once("connect", () => socket.write(`${JSON.stringify({
      token: runtime.token,
      parentSessionId,
      tool: name,
      input,
      context: { callId, lifecycleOwner: "provider" },
    })}\n`))
  })
}
